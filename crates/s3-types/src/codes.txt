# Variant names of `rustfs_s3_types::S3ErrorCode`.
#
# Derived from RustFS's own call surface, never from s3s: every `S3ErrorCode::<Name>`
# path and every `s3_error!(<Name>` invocation in the tree. `Custom` is listed because
# the grep finds it; build.rs emits it as the tuple variant `Custom(Cow<'static, str>)`.
# Names are byte-sorted and unique; build.rs refuses the file otherwise.
#
# Regenerate from the repository root with:
#
#   { git grep -oh 'S3ErrorCode::[A-Za-z0-9]*' -- '*.rs';
#     git grep -oh 's3_error!([A-Za-z0-9]*' -- '*.rs' | sed 's/s3_error!(/S3ErrorCode::/'; } \
#     | sed 's/^S3ErrorCode:://' | grep -E '^[A-Z]' | LC_ALL=C sort -u
#
# The second grep exists because the macro expands to `S3ErrorCode::<Name>`: two codes
# (`MalformedPolicy`, `ServerSideEncryptionConfigurationNotFoundError`) are raised only
# through the macro. `grep -E '^[A-Z]'` drops the `from_bytes` method reference.
